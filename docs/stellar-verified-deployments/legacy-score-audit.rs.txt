const UNIT: i128 = 10_000_000;
const ADMIN: &str = "GCQKVXWIQHVGUOUHZR2O6YBWBNNPKIYH5AWQHXHF3EYRHM73WIWVSK7K";
const PLAYER: &str = "GBGDTBOXVZ4GIDDRISY5FGXYY7XCWCCK3FRHJHSVFEZMWA3ZZ2HUALTK";
const OTHER: &str = "GAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAWHF";

fn address(env: &Env, value: &str) -> Address {
    Address::from_string(&String::from_str(env, value))
}
fn id(env: &Env, value: u8) -> BytesN<32> {
    BytesN::from_array(env, &[value; 32])
}

fn classic_account_with_trustline(env: &Env, who: &Address, asset: &xdr::Asset) {
    let value: soroban_sdk::Val = who.into_val(env);
    let account_id = match xdr::ScVal::try_from_val(env, &value).unwrap() {
        xdr::ScVal::Address(xdr::ScAddress::Account(account)) => account,
        _ => panic!("Expected a classic Stellar account"),
    };
    let account = xdr::AccountEntry {
        account_id: account_id.clone(),
        balance: 1000 * UNIT as i64,
        seq_num: xdr::SequenceNumber(1),
        num_sub_entries: 1,
        inflation_dest: None,
        flags: 0,
        home_domain: Default::default(),
        thresholds: xdr::Thresholds([1, 1, 1, 1]),
        signers: Default::default(),
        ext: xdr::AccountEntryExt::V0,
    };
    env.host()
        .add_ledger_entry(
            &Rc::new(xdr::LedgerKey::Account(xdr::LedgerKeyAccount {
                account_id: account_id.clone(),
            })),
            &Rc::new(xdr::LedgerEntry {
                last_modified_ledger_seq: 100,
                data: xdr::LedgerEntryData::Account(account),
                ext: xdr::LedgerEntryExt::V0,
            }),
            None,
        )
        .unwrap();
    let trust_asset = match asset {
        xdr::Asset::CreditAlphanum4(asset) => xdr::TrustLineAsset::CreditAlphanum4(asset.clone()),
        xdr::Asset::CreditAlphanum12(asset) => xdr::TrustLineAsset::CreditAlphanum12(asset.clone()),
        _ => panic!("Expected an issued asset"),
    };
    let trustline = xdr::TrustLineEntry {
        account_id: account_id.clone(),
        asset: trust_asset.clone(),
        balance: 0,
        limit: i64::MAX,
        flags: 1,
        ext: xdr::TrustLineEntryExt::V0,
    };
    env.host()
        .add_ledger_entry(
            &Rc::new(xdr::LedgerKey::Trustline(xdr::LedgerKeyTrustLine {
                account_id,
                asset: trust_asset,
            })),
            &Rc::new(xdr::LedgerEntry {
                last_modified_ledger_seq: 100,
                data: xdr::LedgerEntryData::Trustline(trustline),
                ext: xdr::LedgerEntryExt::V0,
            }),
            None,
        )
        .unwrap();
}

// Existing public addresses in a simulated host. No signing key or real wallet
// is generated, and no network transaction is submitted by this suite.
fn world() -> (Env, Address, Address, Address, Address, Address) {
    let mut env = Env::default();
    env.set_config(soroban_sdk::testutils::EnvTestConfig {
        capture_snapshot_at_drop: false,
    });
    env.mock_all_auths();
    env.ledger().with_mut(|ledger| ledger.sequence_number = 100);
    let admin = address(&env, ADMIN);
    let p1 = address(&env, PLAYER);
    let p2 = address(&env, OTHER);
    let asset = env.register_stellar_asset_contract_v2(admin.clone());
    let token = asset.address();
    for who in [&admin, &p1, &p2] {
        classic_account_with_trustline(&env, who, &asset.asset());
    }
    let minter = token::StellarAssetClient::new(&env, &token);
    for who in [&admin, &p1, &p2] {
        minter.mint(who, &(50_000 * UNIT));
    }
    let contract = env.register(
        ChainDuel,
        (
            admin.clone(),
            admin.clone(),
            token.clone(),
            1000_u32,
            MAX_ENTRY,
            2 * MAX_ENTRY,
            100 * UNIT,
        ),
    );
    (env, contract, token, admin, p1, p2)
}

fn pvp(env: &Env, contract: &Address, p1: &Address, p2: &Address, key: u8, entry: i128) {
    let client = ChainDuelClient::new(env, contract);
    client.create_game(p1, &id(env, key), &MODE_PVP, &entry, &500);
    client.join_game(p2, &id(env, key));
    client.start_game(&id(env, key));
}

fn settle_p1(env: &Env, contract: &Address, p1: &Address, key: u8) -> Game {
    ChainDuelClient::new(env, contract).settle_game(
        &id(env, key),
        p1,
        &90,
        &10,
        &10,
        &2,
        &id(env, 250),
    )
}

#[test]
fn all_administrative_roles_are_the_exact_designated_wallet() {
    let (env, contract, token, admin, _, _) = world();
    let client = ChainDuelClient::new(&env, &contract);
    let config = client.get_config();
    assert_eq!(config.admin, address(&env, ADMIN));
    for recipient in [
        config.treasury,
        config.protocol_fee_recipient,
        config.ecosystem_fee_recipient,
        config.ecosystem_beneficiary,
    ] {
        assert_eq!(recipient, admin);
    }
    assert_eq!(config.token, token);
    assert_eq!(
        (
            config.fee_bps,
            config.protocol_fee_bps,
            config.ecosystem_fee_bps
        ),
        (1000, 500, 500)
    );
    assert_eq!(client.get_contract_balance(), 0);
}

#[test]
fn a_player_signature_alone_cannot_reserve_an_app_game_id_or_seat() {
    let (env, contract, token, admin, _, _) = world();
    // Selective SDK auth mocks use simulated contract accounts. Classic G
    // account/trustline behavior is covered by all other fixture tests.
    let p1 = Address::generate(&env);
    let p2 = Address::generate(&env);
    token::StellarAssetClient::new(&env, &token).mint(&p1, &(50_000 * UNIT));
    token::StellarAssetClient::new(&env, &token).mint(&p2, &(50_000 * UNIT));
    let client = ChainDuelClient::new(&env, &contract);
    let game_id = id(&env, 33);
    env.mock_auths(&[MockAuth {
        address: &p1,
        invoke: &MockAuthInvoke {
            contract: &contract,
            fn_name: "create_game",
            args: (p1.clone(), game_id.clone(), MODE_PVP, MIN_ENTRY, 500_u32).into_val(&env),
            sub_invokes: &[],
        },
    }]);
    assert!(client
        .try_create_game(&p1, &game_id, &MODE_PVP, &MIN_ENTRY, &500)
        .is_err());
    assert_eq!(token::Client::new(&env, &token).balance(&p1), 50_000 * UNIT);
    env.mock_all_auths();
    assert_eq!(client.get_config().admin, admin);
    client.create_game(&p1, &game_id, &MODE_PVP, &MIN_ENTRY, &500);
    env.mock_auths(&[MockAuth {
        address: &p2,
        invoke: &MockAuthInvoke {
            contract: &contract,
            fn_name: "join_game",
            args: (p2.clone(), game_id.clone()).into_val(&env),
            sub_invokes: &[],
        },
    }]);
    assert!(client.try_join_game(&p2, &game_id).is_err());
    assert_eq!(client.get_locked_escrow(), MIN_ENTRY);
}

#[test]
fn a_100_usdc_pot_pays_90_and_accounts_for_two_independent_fives() {
    let (env, contract, token, _, p1, p2) = world();

    assert_eq!(client.get_accrued_fees(), 10 * UNIT);
    assert_eq!(client.get_locked_escrow(), 0);
    assert_eq!(client.get_contract_balance(), 10 * UNIT);
    assert_eq!(client.get_stats().volume, 100 * UNIT);
}

#[test]
fn withdrawal_ids_cannot_be_replayed_or_reused_for_another_bucket() {
    let (env, contract, token, admin, p1, p2) = world();
    let client = ChainDuelClient::new(&env, &contract);
    pvp(&env, &contract, &p1, &p2, 1, 50 * UNIT);
    settle_p1(&env, &contract, &p1, 1);
    client.withdraw_protocol_fees(&id(&env, 10), &UNIT);
    let before = token::Client::new(&env, &token).balance(&admin);
    let replay_error =
        soroban_sdk::Error::from_contract_error(Error::WithdrawalAlreadyExecuted as u32);
    assert!(
        matches!(client.try_withdraw_protocol_fees(&id(&env, 10), &UNIT), Err(Ok(error)) if error == replay_error)
    );
    assert!(
        matches!(client.try_withdraw_ecosystem_fees(&id(&env, 10), &UNIT), Err(Ok(error)) if error == replay_error)
    );
    assert_eq!(token::Client::new(&env, &token).balance(&admin), before);
    assert_eq!(client.get_protocol_fees(), 4 * UNIT);
    assert_eq!(client.get_ecosystem_fees(), 5 * UNIT);
}

        .try_withdraw_bot_liquidity(&UNIT, &id(&env, 13))

#[test]
fn audit_witness_the_authorized_score_oracle_is_not_a_gameplay_proof() {
    // A passing witness documents a trust boundary; it does NOT demonstrate
    // that a malicious administrator is unable to fabricate a computer win.
    let (env, contract, token, admin, p1, _) = world();
    let client = ChainDuelClient::new(&env, &contract);
    client.fund_bot_pool(&admin, &(125 * UNIT));
    client.create_game(&p1, &id(&env, 93), &MODE_BOT, &MIN_ENTRY, &500);
    client.start_game(&id(&env, 93));
    let before = token::Client::new(&env, &token).balance(&admin);
    let result = client.settle_game(&id(&env, 93), &admin, &0, &1, &0, &0, &id(&env, 94));
    assert_eq!(result.payout, 45 * UNIT);
    assert_eq!(token::Client::new(&env, &token).balance(&admin) - before, 45 * UNIT);
    // No gameplay trace or proof was supplied. The hash is an attestation by
    // the authorized oracle, not evidence the scores came from real gameplay.
}

#[test]
fn varied_256_games_conserve_supply_and_every_withdrawable_bucket() {
    let mut state: u64 = 9137;
    for _ in 0..256 {
        let (env, contract, token, admin, p1, p2) = world();
        let client = ChainDuelClient::new(&env, &contract);
        let balances = token::Client::new(&env, &token);
        state = state.wrapping_mul(6364136223846793005).wrapping_add(1);
        let entry = MIN_ENTRY + (state as i128 % (MAX_ENTRY - MIN_ENTRY + 1));
        if state % 2 == 0 {
            pvp(&env, &contract, &p1, &p2, 1, entry);
        } else {
            client.fund_bot_pool(&admin, &entry);
            client.create_game(&p1, &id(&env, 1), &MODE_BOT, &entry, &500);
            client.start_game(&id(&env, 1));
        }
        let game = settle_p1(&env, &contract, &p1, 1);
        assert_eq!(game.protocol_fee, entry * 2 * 500 / 10000);
        assert_eq!(game.ecosystem_fee, entry * 2 * 500 / 10000);
        assert_eq!(
            game.payout + game.protocol_fee + game.ecosystem_fee,
            entry * 2
        );
        assert_eq!(client.get_locked_escrow(), 0);
        assert_eq!(
            client.get_contract_balance(),
            client.get_bot_liquidity() + client.get_protocol_fees() + client.get_ecosystem_fees()
        );
        client.withdraw_protocol_fees(&id(&env, 10), &game.protocol_fee);
        client.withdraw_ecosystem_fees(&id(&env, 11), &game.ecosystem_fee);
        assert_eq!(
            balances.balance(&admin)
                + balances.balance(&p1)
                + balances.balance(&p2)
                + balances.balance(&contract),
            150_000 * UNIT
        );
    }
}
